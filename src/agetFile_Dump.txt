interface Props {
  step: number;
  total: number;
}

const { step, total } = Astro.props;

const percentage =
  Math.round((step / total) * 100);

const steps = Array.from(
  { length: total },
  (_, i) => i + 1
);

import { updateProgress } from "../../lib/assessment/progress";

const STORAGE_KEY = "ragan_ai_assessment";
const TOTAL_STEPS = 7;
let currentStep = 1;

function getAnswers(): Record<string, any> {
  try {
    return JSON.parse(localStorage.getItem(STORAGE_KEY) ?? "{}");
  } catch {
    return {};
  }
}

function saveSingleAnswer(id: string, value: string) {
  const data = getAnswers();
  data[id] = value;
  localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
}

function toggleMultipleAnswer(id: string, value: string) {
  const data = getAnswers();
  let current: string[] = Array.isArray(data[id]) ? data[id] : [];

  if (current.includes(value)) {
    current = current.filter((item) => item !== value);
  } else {
    current.push(value);
  }

  data[id] = current;
  localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
}

// Validation helper for mandatory questions on the active step
function validateCurrentStep(): boolean {
  const currentStepEl = document.querySelector(`.wizard-step[data-step="${currentStep}"]`);
  if (!currentStepEl) return true;

  const questions = currentStepEl.querySelectorAll(".assessment-question");
  const data = getAnswers();
  let isValid = true;

  questions.forEach((q) => {
    const el = q as HTMLElement;
    const id = el.dataset.questionId;
    const isRequired = el.dataset.required === "true";
    const errorMsg = el.querySelector(".error-msg");

    if (!id || !isRequired) return;

    const val = data[id];
    const isOtherSelected = val === "Other" || (Array.isArray(val) && val.includes("Other"));
    const otherVal = data[`${id}_other`]?.trim();

    // Check missing answer OR missing text when "Other" is selected
    const isMissingAnswer = !val || (Array.isArray(val) && val.length === 0);
    const isMissingOtherText = isOtherSelected && !otherVal;

    if (isMissingAnswer || isMissingOtherText) {
      isValid = false;
      errorMsg?.classList.remove("hidden");
    } else {
      errorMsg?.classList.add("hidden");
    }
  });

  return isValid;
}

// Restores state for both option buttons and "Other" inputs
function syncAssessmentUI() {
  const data = getAnswers();

  document.querySelectorAll(".assessment-question").forEach((q) => {
    const el = q as HTMLElement;
    const id = el.dataset.questionId;
    if (!id) return;

    const val = data[id];
    const options = el.querySelectorAll(".answer-option");
    const otherContainer = el.querySelector(".other-input-container");
    const otherInput = el.querySelector(".other-text-input") as HTMLInputElement | null;

    let isOtherSelected = false;

    options.forEach((opt) => {
      const optEl = opt as HTMLElement;
      const optVal = optEl.dataset.answer;

      const isSelected = Array.isArray(val) ? val.includes(optVal) : val === optVal;
      
      if (isSelected) {
        optEl.classList.add("border-blue-600", "bg-blue-50", "text-blue-700");
      } else {
        optEl.classList.remove("border-blue-600", "bg-blue-50", "text-blue-700");
      }

      if (optVal === "Other" && isSelected) {
        isOtherSelected = true;
      }
    });

    // Toggle and populate "Other" input field
    if (otherContainer) {
      if (isOtherSelected) {
        otherContainer.classList.remove("hidden");
        if (otherInput) {
          otherInput.value = data[`${id}_other`] || "";
        }
      } else {
        otherContainer.classList.add("hidden");
      }
    }
  });
}

function showStep(step: number) {
  document.querySelectorAll(".wizard-step").forEach((section) => {
    section.classList.add("hidden");
  });

  const active = document.querySelector(`.wizard-step[data-step="${step}"]`);
  active?.classList.remove("hidden");

  currentStep = step;
  updateProgress(currentStep, TOTAL_STEPS);
  
  syncAssessmentUI();
  window.scrollTo({ top: 0, behavior: "smooth" });
}

// Handle Navigation (Prevents advancing if required questions fail validation)
document.addEventListener("click", (event) => {
  const target = event.target;
  if (!(target instanceof HTMLElement)) return;

  const button = target.closest("[data-action]");
  if (!button) return;

  const action = (button as HTMLElement).dataset.action;

  if (action === "next" && currentStep < TOTAL_STEPS) {
    if (validateCurrentStep()) {
      showStep(currentStep + 1);
    }
  } else if (action === "back" && currentStep > 1) {
    showStep(currentStep - 1);
  }
});

// Handle Option Button Clicks
document.addEventListener("click", (event) => {
  const target = event.target;
  if (!(target instanceof HTMLElement)) return;

  const option = target.closest(".answer-option") as HTMLElement | null;
  if (!option) return;

  const question = option.closest(".assessment-question") as HTMLElement | null;
  if (!question) return;

  const id = question.dataset.questionId;
  const value = option.dataset.answer;
  const isMultiple = question.dataset.multiple === "true";

  if (!id || !value) return;

  if (isMultiple) {
    toggleMultipleAnswer(id, value);
  } else {
    saveSingleAnswer(id, value);
  }

  // Clear validation message when option clicked
  question.querySelector(".error-msg")?.classList.add("hidden");

  syncAssessmentUI();
});

// Listen for custom "Other" text input changes
document.addEventListener("input", (event) => {
  const target = event.target as HTMLInputElement;

  const id = target.dataset.questionId;
  if (!id) return;

  const data = getAnswers();
  data[`${id}_other`] = target.value;
  localStorage.setItem(STORAGE_KEY, JSON.stringify(data));

  // Hide error message if user provided input
  const question = target.closest(".assessment-question");
  if (target.value.trim().length > 0) {
    question?.querySelector(".error-msg")?.classList.add("hidden");
  }
});

window.addEventListener("syncAssessmentUI", syncAssessmentUI);

<script>
  const menu = document.getElementById("mobileMenu");
  const openBtn = document.getElementById("menuButton");
  const closeBtn = document.getElementById("closeMenu");

  const toggleMenu = (show: boolean) => {
    if (show) {
      menu?.classList.remove("hidden");
      document.body.style.overflow = "hidden";
    } else {
      menu?.classList.add("hidden");
      document.body.style.overflow = "";
    }
  };

  openBtn?.addEventListener("click", () => toggleMenu(true));
  closeBtn?.addEventListener("click", () => toggleMenu(false));

  // Close menu on Escape key press
  document.addEventListener("keydown", (e) => {
    if (e.key === "Escape" && !menu?.classList.contains("hidden")) {
      toggleMenu(false);
    }
  });

  // Close menu when clicking any menu link
  menu?.querySelectorAll("a").forEach((link) => {
    link.addEventListener("click", () => toggleMenu(false));
  });
</script>
